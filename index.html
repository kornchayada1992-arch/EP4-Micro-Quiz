ทำครั้งเดียว ผลทุกคนจะบันทึกอัตโนมัติ

📋 ขั้นตอน:
1. เปิด Google Sheets (ไฟล์เดิม EP.3) → Extensions → Apps Script
2. สร้าง Script ใหม่ หรือเพิ่ม function ใหม่:
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName("ผลทดสอบ EP4");
  if (!sh) {
    sh = ss.insertSheet("ผลทดสอบ EP4");
    sh.appendRow([
      "วันที่-เวลา","ชื่อ-นามสกุล","แผนก","รหัสพนักงาน",
      "คะแนน","คะแนนเต็ม","%","pass100",
      "ข้อ1","ข้อ1✓","ข้อ2","ข้อ2✓",
      "ข้อ3","ข้อ3✓","ข้อ4","ข้อ4✓","ข้อ5","ข้อ5✓"
    ]);
    sh.getRange(1,1,1,18)
      .setFontWeight("bold")
      .setBackground("#0d9e73")
      .setFontColor("white");
  }
  var d = JSON.parse(e.postData.contents);
  sh.appendRow([
    d.datetime, d.name, d.dept, d.empCode,
    d.score, d.total, d.percent, d.pass,
    d.ans1, d.ans1check, d.ans2, d.ans2check,
    d.ans3, d.ans3check, d.ans4, d.ans4check,
    d.ans5, d.ans5check
  ]);
  return ContentService
    .createTextOutput("OK")
    .setMimeType(ContentService.MimeType.TEXT);
}
function doGet() {
  return ContentService
    .createTextOutput("Connected!")
    .setMimeType(ContentService.MimeType.TEXT);
}
